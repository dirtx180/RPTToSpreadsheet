Function FindSelectedListItem(ByVal hList As List, ByVal strSearchItem As String, ByVal strColumnTitle As String) As Integer
Dim nLoop As Integer
Dim nFoundLocation As Integer
Dim strSelectedItem As String

' Initialise variables.
nFoundLocation = 0

' Loop through list items.
For nLoop = 1 To hList.Count
' Determine if current item is selected.
If hList.Selected(nLoop) Then
' Obtain value of selected row in the specified column.
strSelectedItem = ""
strSelectedItem = Trim$(hList.RowItem(nLoop, hList.FindColumn(strColumnTitle)))

' Determine if selected item is same as the search item.
If strSelectedItem = strSearchItem Then
' Match found, hold the row.
nFoundLocation = nLoop

' Stop further processing.
Exit For
End If ' Match Check
End If ' Selected Row Check
Next nLoop

' Return value.
FindSelectedListItem = nFoundLocation
End Function

Function MakeIncludeString(ByVal hList As List, ByVal strArea As String) As String
Const sDELIMITER_COMMA As String = ","

Const sSECTION_TIME_SUMMARY As String = "TS"
Const sSECTION_TIME_DETAIL As String = "TD"
Const sSECTION_DISB_SUMMARY As String = "DS"
Const sSECTION_DISB_DETAIL As String = "DD"
Const sSECTION_DISB_DETAIL_DISB_ONLY As String = "DDDI"

Const sCOLUMN_TITLE_ROW_UNO As String = "Row Uno"
Const sCOLUMN_TITLE_DD_NAME As String = "DD Name"
Const sCOLUMN_TITLE_TIME_FIELD As String = "Time Field"
Const sCOLUMN_TITLE_DISB_FIELD As String = "Disbursement Field"
Const sCOLUMN_TITLE_DISB_CODE As String = "Disbursement Type"
Const sCOLUMN_TITLE_DESCRIPTION As String = "Description"
Const sCOLUMN_TITLE_MIN_SIZE As String = "Min Size"
Const sCOLUMN_TITLE_MAX_SIZE As String = "Max Size"

Dim nRows As Long
Dim nLoop As Long

Dim strRowUno As String
Dim strDDName As String
Dim strField As String
Dim strDesc As String
DIm strMinSize As String
Dim strMaxSize As String

Dim strData As String

' Obtain number of rows in source list.
nRows = hList.Count

strData = ""

' Determine if there are rows in the list.
If nRows > 0 Then
' Loop through list.
For nLoop = 1 To nRows
' Determine if using disbursement detail disbursement code list.
If strArea <> sSECTION_DISB_DETAIL_DISB_ONLY Then
' Not disbursement code list.
' Obtain information from selected row.
strRowUno = hList.RowItem(nLoop, hList.FindColumn(sCOLUMN_TITLE_ROW_UNO))
strDDName = hList.RowItem(nLoop, hList.FindColumn(sCOLUMN_TITLE_DD_NAME))

' Determine the section area.
Select Case strArea
Case sSECTION_TIME_SUMMARY, sSECTION_TIME_DETAIL
strField = hList.RowItem(nLoop, hList.FindColumn(sCOLUMN_TITLE_TIME_FIELD))
Case sSECTION_DISB_SUMMARY, sSECTION_DISB_DETAIL
strField = hList.RowItem(nLoop, hList.FindColumn(sCOLUMN_TITLE_DISB_FIELD))
End Select

strDesc = hList.RowItem(nLoop, hList.FindColumn(sCOLUMN_TITLE_DESCRIPTION))
strMinSize = hList.RowItem(nLoop, hList.FindColumn(sCOLUMN_TITLE_MIN_SIZE))
strMaxSize = hList.RowItem(nLoop, hList.FindColumn(sCOLUMN_TITLE_MAX_SIZE))

' Build row data.
strData = strData + strDDName + sDELIMITER_COMMA
Else
' Disbursement code list.
strDDName = hList.RowItem(nLoop, hList.FindColumn(sCOLUMN_TITLE_DISB_CODE))

' Build row data.
strData = strData + strDDName + sDELIMITER_COMMA
End If
Next nLoop

' Remove trailing delimiter.
strData = Left$(strData, Len(strData) - 1)
End If ' Row Check.

' Return data.
MakeIncludeString = strData
End Function

Function MakeMultiString(ByVal strCode As String, ByVal strDelimiter As String, ByVal strEnclosedChar As String) As String
Dim nPos As Integer
Dim nStart As Integer
Dim strTemp As String

' Initialise variables.
nPos = 0
strTemp = ""

strCode = Trim$(strCode)

' Find initial position of delimiter.
nPos = InStr(strCode, strDelimiter)

' Determine if find initial delimiter.
If nPos = 0 Then
' Single item already selected.
strTemp = strEnclosedChar + strCode + strEnclosedChar
Else
While nPos <> 0
' Build string.
strTemp = strTemp + strEnclosedChar + Left$(strCode, nPos - 1) + strEnclosedChar + strDelimiter

' Rebuild string.
strCode = Trim$(Mid$(strCode, nPos + 1, Len(strCode)))

' Find subsequent position of delimiter.
nPos = InStr(strCode, strDelimiter)
WEnd

' Enclose last item.
strTemp = strTemp + strEnclosedChar + strCode + strEnclosedChar
End If

' Return the string.
MakeMultiString = strTemp
End Function

Dim nRows As Long

Dim strRowUno As String
Dim strDDName As String
Dim strField As String
Dim strDesc As String
DIm strMinSize As String
Dim strMaxSize As String

Dim strData As String

' Obtain number of rows in source list.
nRows = hSourceList.Count

' Loop through source list .
For nLoop = 1 To nRows
' Determine if current source row is selected.
If hSourceList.Selected(nLoop) Then
' Determine if using disbursement detail disbursement code list.
If strArea <> sSECTION_DISB_DETAIL_DISB_ONLY Then
' Not disbursement code list.
' Obtain information from selected row.
strRowUno = hSourceList.RowItem(nLoop, hSourceList.FindColumn(sCOLUMN_TITLE_ROW_UNO))
strDDName = hSourceList.RowItem(nLoop, hSourceList.FindColumn(sCOLUMN_TITLE_DD_NAME))

' Determine the section area.
Select Case strArea
Case sSECTION_TIME_SUMMARY, sSECTION_TIME_DETAIL
strField = hSourceList.RowItem(nLoop, hSourceList.FindColumn(sCOLUMN_TITLE_TIME_FIELD))
Case sSECTION_DISB_SUMMARY, sSECTION_DISB_DETAIL
strField = hSourceList.RowItem(nLoop, hSourceList.FindColumn(sCOLUMN_TITLE_DISB_FIELD))

strDesc = hSourceList.RowItem(nLoop, hSourceList.FindColumn(sCOLUMN_TITLE_DESCRIPTION))
